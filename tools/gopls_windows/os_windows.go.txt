package cache

import (
	"fmt"
	"path/filepath"
	"strings"

	"golang.org/x/sys/windows"
)

func init() {
	checkPathValid = windowsCheckPathValid
}

func windowsCheckPathValid(path string) error {
	name, err := windows.UTF16PtrFromString(path)
	if err != nil {
		return err
	}
	handle, err := windows.CreateFile(name, windows.FILE_READ_ATTRIBUTES,
		windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE|windows.FILE_SHARE_DELETE,
		nil, windows.OPEN_EXISTING, windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
	if err != nil {
		return err
	}
	defer windows.CloseHandle(handle)
	// Query the normalized spelling of the directory already opened
	// under the restricted token, rather than enumerating parents.
	var buf [32768]uint16
	size, err := windows.GetFinalPathNameByHandle(handle, &buf[0], uint32(len(buf)), 0)
	if err != nil {
		return err
	}
	if size == 0 || size >= uint32(len(buf)) {
		return fmt.Errorf("normalized workspace path exceeds bound")
	}
	normalized := windows.UTF16ToString(buf[:size])
	const prefix = "\\\\?\\"
	if !strings.HasPrefix(normalized, prefix) {
		return fmt.Errorf("workspace path is not a normalized drive path")
	}
	normalized = normalized[len(prefix):]
	if !filepath.IsAbs(normalized) || len(filepath.VolumeName(normalized)) != 2 ||
		!strings.EqualFold(filepath.VolumeName(path), filepath.VolumeName(normalized)) ||
		path[2:] != normalized[2:] {
		return fmt.Errorf("invalid path %q: Windows reports %q", path, normalized)
	}
	return nil
}
